char *strrev(char *s)
{
    int i=0,len=strlen(s)-1;
    char temp;
    while(i<len)
    {
        temp=s[i];
        s[i]=s[len];
        s[len]=temp;
        i++;
        len--;
    }
    return s;
}

Simple Linux kernel module to demo spinlock:
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/spinlock.h>
#include <linux/kthread.h>
#include <linux/delay.h>

static spinlock_t mylock;
static int counter;
static struct task_struct *worker_thread;

static int worker_fn(void *data)
{
    while (!kthread_should_stop()) {
        // critical section
        spin_lock(&mylock);
        counter++;
        spin_unlock(&mylock);

        // simulate doing some work
        msleep(10);
    }
    return 0;
}

static int __init spinlock_demo_init(void)
{
    int ret;

    spin_lock_init(&mylock);
    counter = 0;

    worker_thread = kthread_run(worker_fn, NULL, "spin_demo_thread");
    if (IS_ERR(worker_thread)) {
        ret = PTR_ERR(worker_thread);
        pr_err("spinlock_demo: failed to create thread (%d)\n", ret);
        return ret;
    }

    pr_info("spinlock_demo: module loaded\n");
    return 0;
}

static void __exit spinlock_demo_exit(void)
{
    if (worker_thread)
        kthread_stop(worker_thread);

    pr_info("spinlock_demo: module unloaded, final counter = %d\n", counter);
}

module_init(spinlock_demo_init);
module_exit(spinlock_demo_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("NVN");
MODULE_DESCRIPTION("Simple spinlock demo (works on ARM and x86)");

#ifndef XCHG_H
#define XCHG_H

#include <stdint.h>

/*
 * Cross-platform atomic exchange:
 *    old = xchg(addr, newval);
 *
 * Implemented for:
 *   - x86 / x86_64
 *   - ARMv7   (ldrex/strex)
 *   - ARM64   (ldaxr/stlxr)
 *   - RISC-V  (amoswap)
 */

static inline int xchg(volatile int *addr, int newval)
{
    int old;

#if defined(__x86_64__) || defined(__i386__)
    /**************  X86 / X86_64  **************/
    __asm__ volatile(
        "xchg %0, %1"
        : "=r"(old), "+m"(*addr)
        : "0"(newval)
        : "memory"
    );

#elif defined(__aarch64__)
    /**************  ARM64 / AARCH64  **************/
    int tmp;
    __asm__ volatile(
        "1: ldaxr   %w0, [%2]\n"
        "   stlxr   %w1, %w3, [%2]\n"
        "   cbnz    %w1, 1b\n"
        : "=&r"(old), "=&r"(tmp)
        : "r"(addr), "r"(newval)
        : "memory"
    );

#elif defined(__arm__)
    /**************  ARMv7 (32-bit ARM) **************/
    int tmp;
    __asm__ volatile(
        "1: ldrex   %0, [%2]\n"
        "   strex   %1, %3, [%2]\n"
        "   teq     %1, #0\n"
        "   bne     1b\n"
        : "=&r"(old), "=&r"(tmp)
        : "r"(addr), "r"(newval)
        : "memory"
    );

#elif defined(__riscv)
    /**************  RISC-V (RV32/RV64) **************/
    __asm__ volatile(
        "amoswap.w.aq %0, %2, %1"
        : "=r"(old), "+A"(*addr)
        : "r"(newval)
    );

#else
#error "xchg() not implemented for this architecture"
#endif

    return old;
}

#endif /* XCHG_H */


/*****************  MOCK sleep & wakeup  *****************/
// These are placeholders — in OS code, these block/wake threads.
void sleep(void);
void wakeup(void);


/*********************************************************
 *                     MUTEX (blocking)
 *********************************************************/

typedef struct {
    volatile int locked;   // 0 = free, 1 = locked
} mutex_t;

void mutex_init(mutex_t *m)
{
    m->locked = 0;
}

void mutex_lock(mutex_t *m)
{
    while (1) {
        if (xchg(&m->locked, 1) == 0)  // acquired
            break;
        sleep();                       // block instead of spinning
    }
}

void mutex_unlock(mutex_t *m)
{
    m->locked = 0;
    wakeup();       // wake sleeping waiters
}


/*********************************************************
 *               SPINLOCK (pure spin + no sleep)
 *********************************************************/

typedef struct {
    volatile int locked;  // 0 = free, 1 = locked
} spinlock_t;

void spinlock_init(spinlock_t *s)
{
    s->locked = 0;
}

void spin_lock(spinlock_t *s)
{
    while (xchg(&s->locked, 1) == 1) {
        // busy-wait (spin)
    }
}

void spin_unlock(spinlock_t *s)
{
    s->locked = 0;
}


/*********************************************************
 *            BINARY SEMAPHORE (sleep + wakeup)
 *********************************************************/

typedef struct {
    volatile int count;   // 0 or 1 for binary semaphore
} binsem_t;

void binsem_init(binsem_t *s, int init_val)
{
    s->count = init_val;   // should be 0 or 1
}

void binsem_wait(binsem_t *s)
{
    while (1) {
        // Attempt atomic decrement
        if (xchg(&s->count, 0) == 1) {
            // successfully took semaphore
            break;
        }
        sleep();  // wait until semaphore becomes available
    }
}

void binsem_signal(binsem_t *s)
{
    s->count = 1;
    wakeup();  // wake waiting threads
}
